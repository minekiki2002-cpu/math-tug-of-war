        return `<button class="key" data-side="${side}" data-value="${key}">${key}</button>`;
      }).join("");

      const handleKeyPress = (event) => {
        const button = event.target.closest("button");
        if (!button || !state.active || state.locks[side]) return;
        event.preventDefault();
        handleTeamInput(side, button.dataset.action || button.dataset.value);
      };

      pad.addEventListener("pointerdown", (event) => {
        const button = event.target.closest("button");
        if (button) button.dataset.lastPointer = String(Date.now());
        state.keyboardSide = side;
        handleKeyPress(event);
      });

      pad.addEventListener("click", (event) => {
        const button = event.target.closest("button");
        const lastPointer = Number(button?.dataset.lastPointer || 0);
        if (Date.now() - lastPointer < 500) {
          event.preventDefault();
          return;
        }
        handleKeyPress(event);
      });
    }

    async function toggleFullscreen() {
      const root = document.documentElement;
      const requestFullscreen =
        root.requestFullscreen ||
        root.webkitRequestFullscreen ||
        root.msRequestFullscreen;
      const exitFullscreen =
        document.exitFullscreen ||
        document.webkitExitFullscreen ||
        document.msExitFullscreen;
      const fullscreenElement =
        document.fullscreenElement ||
        document.webkitFullscreenElement ||
        document.msFullscreenElement;

      try {
        if (!fullscreenElement && requestFullscreen) {
          await requestFullscreen.call(root);
          document.body.classList.add("app-fullscreen");
          state.pseudoFullscreen = false;
          return;
        }
        if (fullscreenElement && exitFullscreen) {
          await exitFullscreen.call(document);
          document.body.classList.remove("app-fullscreen");
          state.pseudoFullscreen = false;
          return;
        }
      } catch (error) {
        // Embedded pages often block the Fullscreen API; use CSS fullscreen below.
      }

      state.pseudoFullscreen = !state.pseudoFullscreen;
      document.body.classList.toggle("app-fullscreen", state.pseudoFullscreen);
    }

    function syncFullscreenClass() {
      const fullscreenElement =
        document.fullscreenElement ||
        document.webkitFullscreenElement ||
        document.msFullscreenElement;

      if (fullscreenElement) {
        document.body.classList.add("app-fullscreen");
        state.pseudoFullscreen = false;
      } else if (!state.pseudoFullscreen) {
        document.body.classList.remove("app-fullscreen");
      }
    }

    function bindKeyboardControls() {
      document.addEventListener("keydown", (event) => {
        if (!state.active || event.repeat) return;

        const target = event.target;
        if (target && ["INPUT", "TEXTAREA", "SELECT"].includes(target.tagName) && !target.classList.contains("answer")) {
          return;
        }

        const action = getKeyboardAction(event);
        if (!action) return;
        event.preventDefault();
        handleTeamInput(action.side, action.value);
      });
    }

    function getKeyboardAction(event) {
      if (/^Digit[0-9]$/.test(event.code)) {
        return { side: event.shiftKey || event.altKey ? "right" : state.keyboardSide, value: event.code.replace("Digit", "") };
      }

      if (/^Numpad[0-9]$/.test(event.code)) {
        return { side: "right", value: event.code.replace("Numpad", "") };
      }

      if (event.code === "Backspace") return { side: state.keyboardSide, value: "clear" };
      if (event.code === "Delete") return { side: "right", value: "clear" };
      if (event.code === "Enter") return { side: event.shiftKey || event.altKey ? "right" : state.keyboardSide, value: "submit" };
      if (event.code === "NumpadEnter") return { side: "right", value: "submit" };
      return null;
    }

    function handleTeamInput(side, value) {
      if (!state.active || state.locks[side]) return;
      if (value === "clear") clearInput(side);
      else if (value === "submit") checkAnswer(side);
      else pressKey(side, value);
    }

    function updateSetupView() {
      els.pages.forEach((page) => page.classList.toggle("active", Number(page.dataset.step) === state.setupStep));
      els.stepDots.forEach((dot) => {
        const step = Number(dot.dataset.stepDot);
        dot.classList.toggle("active", step === state.setupStep);
        dot.classList.toggle("done", step < state.setupStep);
      });
      els.back.disabled = state.setupStep === 0;
      els.next.style.display = state.setupStep < 2 ? "inline-flex" : "none";
      els.start.style.display = state.setupStep === 2 ? "inline-flex" : "none";
      els.next.disabled = state.setupStep === 0 && state.selectedOperations.size === 0;
    }

    function startMusic() {
      prepareAudioContext();
      if (!state.audio.context) return;
      if (state.audio.timer) return;
      scheduleMusicLoop();
    }

    function prepareAudioContext() {
      const AudioContext = window.AudioContext || window.webkitAudioContext;
      if (!AudioContext) return;

      if (!state.audio.context) {
        state.audio.context = new AudioContext();
        state.audio.gain = state.audio.context.createGain();
        state.audio.gain.gain.value = 0.12;
        state.audio.gain.connect(state.audio.context.destination);
      }

      if (state.audio.context.state === "suspended") {
        state.audio.context.resume();
      }
    }

    function scheduleMusicLoop() {
      if (state.audio.timer) {
        clearInterval(state.audio.timer);
        state.audio.timer = null;
      }

      const horn = [392, 523, 587, 659, 587, 523, 784, 659];
      state.audio.timer = setInterval(() => {
        const ctx = state.audio.context;
        const now = ctx.currentTime;
        const step = state.audio.step;
        const energy = Math.max(0, GAME_CONFIG.music.startIntervalMs - state.audio.intervalMs);
        const drumBoost = Math.min(0.08, energy / 2200);
        const beat = step % 16;

        if ([0, 4, 8, 12].includes(beat)) playKickAt(0.18 + drumBoost);
        if ([2, 6, 10, 14].includes(beat)) playTaikoAt(154, 0.12 + drumBoost);
        if ([3, 7, 11, 15].includes(beat)) playTaikoAt(218, 0.075 + drumBoost);
        if ([1, 5, 9, 13].includes(beat)) playStickAt(0.034);
        if ([0, 6, 10].includes(beat)) playHornAt(horn[(step + Math.floor(step / 4)) % horn.length], 0.055);
        if (state.audio.intervalMs < 132 && [5, 15].includes(beat)) playShoutAt(0.052);
        state.audio.step += 1;

        function playKickAt(volume) {
          const osc = ctx.createOscillator();
          const gain = ctx.createGain();
          osc.type = "sine";
          osc.frequency.setValueAtTime(108, now);
          osc.frequency.exponentialRampToValueAtTime(48, now + 0.16);
          gain.gain.setValueAtTime(0.0001, now);
          gain.gain.exponentialRampToValueAtTime(volume, now + 0.008);
          gain.gain.exponentialRampToValueAtTime(0.0001, now + 0.18);
          osc.connect(gain);
          gain.connect(state.audio.gain);
          osc.start(now);
          osc.stop(now + 0.2);
        }

        function playTaikoAt(frequency, volume) {
          const osc = ctx.createOscillator();
          const overtone = ctx.createOscillator();
          const gain = ctx.createGain();
          osc.type = "sine";
          overtone.type = "triangle";
          osc.frequency.setValueAtTime(frequency, now);
          osc.frequency.exponentialRampToValueAtTime(frequency * 0.68, now + 0.12);
          overtone.frequency.setValueAtTime(frequency * 1.72, now);
          gain.gain.setValueAtTime(0.0001, now);
          gain.gain.exponentialRampToValueAtTime(volume, now + 0.006);
          gain.gain.exponentialRampToValueAtTime(0.0001, now + 0.14);
          osc.connect(gain);
          overtone.connect(gain);
          gain.connect(state.audio.gain);
          osc.start(now);
          overtone.start(now);
          osc.stop(now + 0.16);
          overtone.stop(now + 0.12);
          playNoiseAt(0.055, volume * 0.22);
        }

        function playHornAt(frequency, volume) {
          const osc = ctx.createOscillator();
          const gain = ctx.createGain();
          osc.type = "sawtooth";
          osc.frequency.value = frequency;
          gain.gain.setValueAtTime(0.0001, now);
          gain.gain.exponentialRampToValueAtTime(volume, now + 0.012);
          gain.gain.exponentialRampToValueAtTime(0.0001, now + 0.075);
          osc.connect(gain);
          gain.connect(state.audio.gain);
          osc.start(now);
          osc.stop(now + 0.095);
        }

        function playStickAt(volume) {
          playNoiseAt(0.024, volume);
        }

        function playShoutAt(volume) {
          playNoiseAt(0.07, volume);
          playHornAt(880, volume * 0.7);
        }

        function playNoiseAt(duration, volume) {
          const buffer = ctx.createBuffer(1, Math.floor(ctx.sampleRate * duration), ctx.sampleRate);
          const data = buffer.getChannelData(0);
          for (let i = 0; i < data.length; i += 1) data[i] = Math.random() * 2 - 1;
          const source = ctx.createBufferSource();
          const gain = ctx.createGain();
          source.buffer = buffer;
          gain.gain.setValueAtTime(volume, now);
          gain.gain.exponentialRampToValueAtTime(0.0001, now + duration);
          source.connect(gain);
          gain.connect(state.audio.gain);
          source.start(now);
        }
      }, state.audio.intervalMs);
    }

    function stopMusic() {
      if (state.audio.timer) {
        clearInterval(state.audio.timer);
        state.audio.timer = null;
      }
    }

    function resetMusicTempo() {
      state.audio.step = 0;
      state.audio.intervalMs = GAME_CONFIG.music.startIntervalMs;
      if (state.audio.timer) scheduleMusicLoop();
    }

    function speedUpMusic() {
      if (!state.musicEnabled) return;
      const totalCorrect = state.scores.left + state.scores.right;
      if (totalCorrect < 4) return;

      const nextInterval = Math.max(
        GAME_CONFIG.music.minIntervalMs,
        state.audio.intervalMs - GAME_CONFIG.music.speedUpMs
      );

      if (nextInterval !== state.audio.intervalMs) {
        state.audio.intervalMs = nextInterval;
        if (state.audio.timer) scheduleMusicLoop();
      }
    }

    function startTimer() {
      state.timerId = setInterval(() => {
        if (!state.active) return;

        updateTimer();
      }, 150);
    }

    function updateTimer(forceZero = false) {
      const totalSeconds = Math.floor((forceZero ? 0 : state.elapsed) / 1000);
      const minutes = String(Math.floor(totalSeconds / 60)).padStart(2, "0");
      const seconds = String(totalSeconds % 60).padStart(2, "0");
      els.timer.textContent = `${minutes}:${seconds}`;
    }

    function generateProblem(side) {
      const ops = Array.from(state.selectedOperations);
      const questionNumber = state.scores[side] + 1;
      const rules = getQuestionRules(questionNumber);
      let problem = null;

      for (let attempt = 0; attempt < 120; attempt += 1) {
        const op = randomChoice(ops) || "+";
        const pattern = randomChoice(rules.patterns);
        const candidate = makeProblem(op, pattern, rules.range);
        if (isProblemAllowed(candidate, rules)) {
          problem = candidate;
          break;
        }
      }

      if (!problem) {
        const op = randomChoice(ops) || "+";
        problem = makeProblem(op, randomChoice(rules.patterns), rules.range);
      }

      state.answers[side] = problem.answer;
      $(`#${side}Problem`).innerHTML = problem.text;
    }

    function getQuestionRules(questionNumber) {
      const progression = GAME_CONFIG.progression[state.difficulty];
      if (questionNumber <= 2) {
        return {
          range: progression.warmupRange,
          patterns: ["standard", "missingLeft", "missingRight"],
          maxAnswer: progression.warmupMaxAnswer,
          minAnswer: 1
        };
      }

      return {
        range: GAME_CONFIG.ranges[state.difficulty],
        patterns: questionNumber >= 4 ? ["missingLeft", "missingRight"] : ["standard", "missingLeft", "missingRight"],
        minAnswer: progression.laterMinAnswer,
        maxAnswer: Infinity
      };
    }

    function isProblemAllowed(problem, rules) {
      return problem.answer >= rules.minAnswer && problem.answer <= rules.maxAnswer;
    }

    function makeProblem(op, pattern, range) {
      if (op === "inverse") return makeInverseProblem();
      if (op === "+") return makeAddProblem(pattern, range);
      if (op === "-") return makeSubProblem(pattern, range);
      if (op === "*") return makeMulProblem(pattern, range);
      return makeDivProblem(pattern, range);
    }

    function makeInverseProblem() {
      const range = GAME_CONFIG.inverseRanges[state.difficulty];
      const a = randomInt(range.aMin, range.aMax);
      const b = randomInt(range.bMin, range.bMax);
      const c = a * b;
      const templates = [
        { text: `${a} × ${b} = ${c}<br>${c} ÷ ${b} = ( )`, answer: a },
        { text: `${a} × ${b} = ${c}<br>${c} ÷ ( ) = ${a}`, answer: b },
        { text: `${c} ÷ ${b} = ${a}<br>${a} × ${b} = ( )`, answer: c },
        { text: `${c} ÷ ${a} = ${b}<br>( ) × ${a} = ${c}`, answer: b },
        { text: `${c} ÷ ${b} = ${a}<br>( ) × ${b} = ${c}`, answer: a },
        { text: `${a} × ${b} = ${c}<br>( ) ÷ ${a} = ${b}`, answer: c }
      ];
      return randomChoice(templates);
    }

    function makeAddProblem(pattern, range) {
      const a = randomInt(range.min, range.max);
      const b = randomInt(range.min, range.max);
      const sum = a + b;
      if (pattern === "missingLeft") return { text: `( ) + ${b} = ${sum}`, answer: a };
      if (pattern === "missingRight") return { text: `${a} + ( ) = ${sum}`, answer: b };
      return { text: `${a} + ${b} = ?`, answer: sum };
    }

    function makeSubProblem(pattern, range) {
      const difference = randomInt(range.min, range.max);
      const subtrahend = randomInt(range.min, range.max);
      const minuend = difference + subtrahend;
      if (pattern === "missingLeft") return { text: `( ) − ${subtrahend} = ${difference}`, answer: minuend };

      const a = randomInt(range.min + 1, range.max + range.min);
      const b = randomInt(range.min, Math.max(range.min, a - 1));
      if (pattern === "missingRight") return { text: `${a} − ( ) = ${a - b}`, answer: b };
      return { text: `${a} − ${b} = ?`, answer: a - b };
    }

    function makeMulProblem(pattern, range) {
      const a = randomInt(range.min, range.max);
      const b = randomInt(range.min, range.max);
      const product = a * b;
      if (pattern === "missingLeft") return { text: `( ) × ${b} = ${product}`, answer: a };
      if (pattern === "missingRight") return { text: `${a} × ( ) = ${product}`, answer: b };
      return { text: `${a} × ${b} = ?`, answer: product };
    }

    function makeDivProblem(pattern, range) {
      const divisorMin = Math.max(2, Math.min(range.min, 12));
      const divisorMax = Math.max(divisorMin, Math.min(range.max, 12));
      const divisor = randomInt(divisorMin, divisorMax);
      const quotient = randomInt(1, range.max);
      const dividend = divisor * quotient;
      if (pattern === "missingLeft") return { text: `( ) ÷ ${divisor} = ${quotient}`, answer: dividend };

      const answerDivisor = randomInt(divisorMin, divisorMax);
      const answerQuotient = randomInt(1, range.max);
      const knownDividend = answerDivisor * answerQuotient;
      if (pattern === "missingRight") return { text: `${knownDividend} ÷ ( ) = ${answerQuotient}`, answer: answerDivisor };
      return { text: `${dividend} ÷ ${divisor} = ?`, answer: quotient };
    }

    function randomInt(min, max) {
      return Math.floor(Math.random() * (max - min + 1)) + min;
    }

    function randomChoice(items) {
      return items[Math.floor(Math.random() * items.length)];
    }

    function pressKey(side, value) {
      const input = $(`#${side}Input`);
      if (input.value.length < GAME_CONFIG.maxInputLength) input.value += value;
    }

    function clearInput(side) {
      $(`#${side}Input`).value = "";
    }

    function checkAnswer(side) {
      const input = $(`#${side}Input`);
      const value = input.value.trim() === "" ? NaN : parseInt(input.value, 10);
      state.locks[side] = true;
      setInputsEnabled(true);

      if (!Number.isNaN(value) && value === state.answers[side]) {
        input.classList.add("correct");
        state.scores[side] += 1;
        speedUpMusic();
        state.position += side === "left" ? -GAME_CONFIG.step : GAME_CONFIG.step;
        updateScores();
        updateRope();
        setTimeout(() => {
          input.classList.remove("correct");
          input.value = "";
          state.locks[side] = false;
          setInputsEnabled(true);
          checkWin();
          if (state.active) generateProblem(side);
        }, 400);
      } else {
        input.classList.add("incorrect", "shake");
        const backStep = GAME_CONFIG.step * GAME_CONFIG.wrongStepFactor;
        state.position += side === "left" ? backStep : -backStep;
        updateRope();
        setTimeout(() => {
          input.classList.remove("incorrect", "shake");
          input.value = "";
          state.locks[side] = false;
          setInputsEnabled(true);
          checkWin();
        }, 500);
      }
    }

    function setInputsEnabled(enabled) {
      ["left", "right"].forEach((side) => {
        const available = enabled && state.active && !state.locks[side];
        $$(`button[data-side="${side}"]`).forEach((button) => {
          button.disabled = !available;
        });
        $(`#${side}Input`).disabled = !state.active;
      });
    }

    function updateScores() {
      $("#leftScore").textContent = state.scores.left;
      $("#rightScore").textContent = state.scores.right;
      $("#leftPanelScore").textContent = state.scores.left;
      $("#rightPanelScore").textContent = state.scores.right;
    }

    function updateRope() {
      const visibleOffset = Math.max(-250, Math.min(250, state.position));
      const sceneOffset = Math.round(visibleOffset * 0.72);
      els.arena.style.setProperty("--pull-x", `${sceneOffset}px`);
      if (state.position === 0) {
        els.distance.textContent = "繩子：中央";
      } else {
        const side = state.position < 0 ? state.names.left : state.names.right;
        els.distance.textContent = `繩子偏向：${side} ${Math.abs(Math.round(state.position))}`;
      }
    }

    function checkWin() {
      if (Math.abs(state.position) >= GAME_CONFIG.winLimit) {
        endGame(state.position < 0 ? "left" : "right");
        return;
      }
      if (state.scores.left >= GAME_CONFIG.winByScore) {
        endGame("left");
        return;
      }
      if (state.scores.right >= GAME_CONFIG.winByScore) {
        endGame("right");
      }
    }

    function endGame(winnerSide) {
      state.active = false;
      clearInterval(state.timerId);
      setInputsEnabled(false);
      els.arena.classList.remove("playing");
      els.arena.classList.add("finished", winnerSide === "left" ? "winner-left" : "winner-right");
      stopMusic();
      playVictoryFanfare();
      $("#winnerName").textContent = state.names[winnerSide];
      $("#winnerMessage").textContent = `${state.names[winnerSide]} 撒花歡慶獲勝！`;
      $("#winnerScore").textContent = state.scores[winnerSide];
      $("#winnerTime").textContent = els.timer.textContent;
      burstConfetti();
      burstFlowers();
      setTimeout(() => {
        els.winner.style.display = "grid";
      }, 5000);
    }

    function burstConfetti() {
      const colors = ["#1769e8", "#e5484d", "#f59e0b", "#10b981", "#8b5cf6"];
      for (let i = 0; i < 60; i += 1) {
        setTimeout(() => {
          const piece = document.createElement("div");

    function burstFlowers() {
      const flowers = ["✿", "✽", "✦", "❀", "✹"];
      for (let i = 0; i < 90; i += 1) {

    function playVictoryFanfare() {
      if (!state.musicEnabled || !state.audio.context || !state.audio.gain) return;
      const ctx = state.audio.context;
      const notes = [523, 659, 784, 1046, 784, 1046];
      notes.forEach((note, index) => {
        const start = ctx.currentTime + index * 0.13;
        const osc = ctx.createOscillator();
        const gain = ctx.createGain();
        osc.type = "triangle";
        osc.frequency.value = note;
        gain.gain.setValueAtTime(0.0001, start);
        gain.gain.exponentialRampToValueAtTime(0.12, start + 0.018);
        gain.gain.exponentialRampToValueAtTime(0.0001, start + 0.18);
        osc.connect(gain);
        gain.connect(state.audio.gain);
        osc.start(start);
        osc.stop(start + 0.22);
      });
    }
